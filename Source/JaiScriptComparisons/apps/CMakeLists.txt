# The general benchmark apps (Richards, DeltaBlue, a bytecode VM, a spatial-hash simulation) run as plain scripts: the
# JaiScript runner, CPython and the two stand-alone interpreters built here (opt-in; built from source).
option(APPS_PEERS "Build apps_lua (Lua 5.4 lua.c) and apps_sq (Squirrel 3.2 sq.c) for the benchmark apps" OFF)
set(APPS_LUA_ROOT "${COMPARISONS_THIRD_PARTY}/lua" CACHE PATH "Lua 5.4 source root (the folder holding src/lua.h)")
set(APPS_SQUIRREL_ROOT "${COMPARISONS_THIRD_PARTY}/squirrel" CACHE PATH "Squirrel 3.2 source root (the folder holding include/squirrel.h)")
if(APPS_PEERS)
	if(NOT EXISTS "${APPS_LUA_ROOT}/src/lua.h" OR NOT EXISTS "${APPS_SQUIRREL_ROOT}/include/squirrel.h")
		message(FATAL_ERROR "APPS_PEERS=ON but Lua/Squirrel not found at ${APPS_LUA_ROOT} / ${APPS_SQUIRREL_ROOT}")
	endif()
	file(GLOB apps_lua_sources "${APPS_LUA_ROOT}/src/*.c")
	list(FILTER apps_lua_sources EXCLUDE REGEX "/(luac|onelua)\\.c$")
	add_executable(apps_lua ${apps_lua_sources})
	target_include_directories(apps_lua PRIVATE "${APPS_LUA_ROOT}/src")
	if(UNIX)
		target_link_libraries(apps_lua PRIVATE m ${CMAKE_DL_LIBS})
	endif()

	file(GLOB apps_sq_sources "${APPS_SQUIRREL_ROOT}/squirrel/*.cpp" "${APPS_SQUIRREL_ROOT}/sqstdlib/*.cpp")
	add_executable(apps_sq "${APPS_SQUIRREL_ROOT}/sq/sq.c" ${apps_sq_sources})
	target_include_directories(apps_sq PRIVATE "${APPS_SQUIRREL_ROOT}/include")
	target_compile_definitions(apps_sq PRIVATE SQUSEDOUBLE)
	if(MSVC)
		target_compile_options(apps_lua PRIVATE /W3)
		target_compile_options(apps_sq PRIVATE /W2)
	endif()
	message(STATUS "apps: apps_lua and apps_sq enabled (${APPS_LUA_ROOT} + ${APPS_SQUIRREL_ROOT})")
endif()
