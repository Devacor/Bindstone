cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
	project(JaiDoomLua LANGUAGES C CXX)
	set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
if(NOT DEFINED COMPARISONS_THIRD_PARTY)
	set(COMPARISONS_THIRD_PARTY "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party")
endif()
if(NOT DEFINED JAISCRIPT_DIR)
	set(JAISCRIPT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../JaiScript")
endif()
set(JAIDOOM_LUA_ROOT "${COMPARISONS_THIRD_PARTY}/lua" CACHE PATH "Lua 5.4/5.5 source root")
set(JAIDOOM_GAME_ROOT "${JAISCRIPT_DIR}/examples/jaidoom" CACHE PATH "JaiDoom game data root")
option(JAIDOOM_PADDING_CONTROLS "Build cold-help-function layout control" ON)
if(TARGET gloom_lua_runtime)
	set(jaidoom_runtime gloom_lua_runtime)
else()
	enable_language(C)
	if(NOT EXISTS "${JAIDOOM_LUA_ROOT}/src/lua.h")
		message(FATAL_ERROR "JAIDOOM_LUA_ROOT must name the pinned Lua source directory")
	endif()
	file(GLOB lua_sources "${JAIDOOM_LUA_ROOT}/src/*.c")
	list(FILTER lua_sources EXCLUDE REGEX "/(lua|luac)\\.c$")
	add_library(jaidoom_lua_runtime STATIC ${lua_sources})
	target_include_directories(jaidoom_lua_runtime PUBLIC "${JAIDOOM_LUA_ROOT}/src")
	if(MSVC)
		set_property(TARGET jaidoom_lua_runtime PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
	elseif(UNIX)
		target_link_libraries(jaidoom_lua_runtime PUBLIC m ${CMAKE_DL_LIBS})
	endif()
	set(jaidoom_runtime jaidoom_lua_runtime)
endif()
function(jaidoom_lua_target target)
	add_executable(${target} lua_host.cpp)
	target_compile_features(${target} PRIVATE cxx_std_20)
	target_link_libraries(${target} PRIVATE ${jaidoom_runtime})
	target_compile_definitions(${target} PRIVATE JAIDOOM_LUA_SCRIPTS="${CMAKE_CURRENT_SOURCE_DIR}/lua" JAIDOOM_GAME_ROOT="${JAIDOOM_GAME_ROOT}")
	if(MSVC)
		target_compile_options(${target} PRIVATE /W3 /EHsc /bigobj $<$<CONFIG:Release>:/O2;/Ob2;/Oi;/Ot;/GL>)
		target_link_options(${target} PRIVATE $<$<CONFIG:Release>:/LTCG;/OPT:REF;/OPT:ICF>)
		set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
	endif()
endfunction()
jaidoom_lua_target(jai_doom_lua)
if(JAIDOOM_PADDING_CONTROLS)
	jaidoom_lua_target(jai_doom_lua_padding)
	target_compile_definitions(jai_doom_lua_padding PRIVATE JAIDOOM_LAYOUT_PADDING=1)
endif()

option(JAIDOOM_PORT_TOOLS "Build JaiScript frame oracle and AST export tool" OFF)
if(JAIDOOM_PORT_TOOLS)
	if(NOT TARGET jaiscript)
		message(FATAL_ERROR "JAIDOOM_PORT_TOOLS requires the main JaiScript project")
	endif()
	add_executable(jai_doom_reference reference_host.cpp)
	add_executable(jai_doom_export_ast tools/export_ast.cpp)
	foreach(tool IN ITEMS jai_doom_reference jai_doom_export_ast)
		target_compile_features(${tool} PRIVATE cxx_std_20)
		target_link_libraries(${tool} PRIVATE jaiscript)
		if(MSVC)
			target_compile_options(${tool} PRIVATE /bigobj /EHsc)
			set_property(TARGET ${tool} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
		endif()
	endforeach()
endif()
